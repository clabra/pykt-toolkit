\documentclass[tikz, border=20pt]{standalone}
\usepackage{tikz}
\usetikzlibrary{positioning, arrows.meta, shapes, fit, calc}

\begin{document}

\definecolor{input_color}{RGB}{245,245,245}
\definecolor{emb_color}{RGB}{224,242,252}
\definecolor{grounding_color}{RGB}{255,224,224}
\definecolor{attn_color}{RGB}{242,243,193}
\definecolor{ff_color}{RGB}{194,232,247}
\definecolor{output_color}{RGB}{220,223,240}
\definecolor{loss_color}{RGB}{255,245,230}
\definecolor{bkt_color}{RGB}{240,240,240}

\begin{tikzpicture}[
    node distance=1.2cm and 1.2cm,
    stage_box/.style={draw, gray!50, dashed, rounded corners=8pt, line width=1pt},
    stage_label/.style={font=\bfseries\Large, gray!60!black, anchor=south west, inner sep=10pt},
    block/.style={draw, rectangle, rounded corners=3pt, minimum width=2.4cm, minimum height=1.1cm, align=center, fill=white, line width=0.7pt, font=\small},
    data/.style={block, fill=input_color},
    embedding/.style={block, fill=emb_color},
    grounding/.style={block, fill=grounding_color},
    core/.style={block, fill=attn_color},
    output/.style={block, fill=output_color},
    loss_node/.style={block, fill=loss_color, minimum width=1.8cm},
    ref_node/.style={block, fill=bkt_color, dashed},
    arrow/.style={-{Stealth[scale=1.2]}, thick, gray!50!black},
    path_line/.style={thick, gray!40!black},
    comp_label/.style={font=\itshape\footnotesize, gray!60!black}
]

    % ==========================================
    % LAYER 1: Input Data (BOTTOM)
    % ==========================================
    \node[data] (I_q) {Questions $q$};
    \node[data, right=of I_q] (I_r) {Responses $r$};
    \node[data, right=of I_r] (I_pid) {Problem IDs $pid$};
    \node[data, right=of I_pid] (I_uid) {Student IDs $uid$};
    \node[ref_node, right=2cm of I_uid] (I_bkt) {BKT Reference\\Targets};

    \node[stage_box, fit=(I_q) (I_bkt), inner sep=20pt] (Stage1) {};
    \node[stage_label] at (Stage1.south west) {1. Input Data};

    % ==========================================
    % LAYER 2: Embeddings
    % ==========================================
    \node[embedding, above=3.5cm of I_q] (E_q) {Concept\\Embeddings};
    \node[embedding, above=3.5cm of I_r] (E_qa) {Interaction\\Embeddings};
    \node[grounding, above=3.5cm of I_pid] (G_uq) {Question\\Difficulty ($u_q$)};
    \node[grounding, above=3.5cm of I_uid] (G_st) {Student\\Params ($k_c, v_s$)};

    \node[grounding, above=1.2cm of E_q] (X_prime) {Transition Gap\\Task $x'$};
    \node[grounding, above=1.2cm of E_qa] (Y_prime) {Transition Gain\\History $y'$};

    \node[stage_box, fit=(E_q) (G_st) (X_prime) (Y_prime), inner sep=20pt] (Stage2) {};
    \node[stage_label] at (Stage2.south west) {2. Embeddings (Representational Grounding)};

    % ==========================================
    % LAYER 3: Transformer Core
    % ==========================================
    \node[core, above=3cm of X_prime] (Decoder) {Decoder\\(2N blocks)};
    \node[comp_label, below=0.1cm of Decoder] {Self \& Cross-Attn};

    \node[core, above=3cm of Y_prime] (Encoder) {Encoder\\(N blocks)};
    \node[comp_label, below=0.1cm of Encoder] {Self-Attn on $y'$};

    \node[stage_box, fit=(Encoder) (Decoder), inner sep=20pt] (Stage3) {};
    \node[stage_label] at (Stage3.south west) {3. Transformer Core};

    % ==========================================
    % LAYER 4: Output Stage
    % ==========================================
    \node[output, above=3cm of Decoder] (MLP) {MLP Stage\\(Linear Layers)};
    \node[output, above=1cm of MLP] (Predict) {iDKT Prediction\\($p_{iDKT}$)};

    \node[stage_box, fit=(MLP) (Predict), inner sep=20pt] (Stage4) {};
    \node[stage_label] at (Stage4.south west) {4. Output Stage};

    % ==========================================
    % LAYER 5: Loss Functions (TOP)
    % ==========================================
    \node[loss_node, above=4cm of Predict] (L_sup) {$L_{sup}$};
    \node[loss_node, right=0.6cm of L_sup] (L_ref) {$L_{ref}$};
    \node[loss_node, right=0.6cm of L_ref] (L_init) {$L_{init}$};
    \node[loss_node, right=0.6cm of L_init] (L_rate) {$L_{rate}$};
    \node[loss_node, right=0.6cm of L_rate] (L_rasch) {$L_{rasch}$};
    \node[loss_node, right=0.6cm of L_rasch] (L_reg) {$L_{reg}$};
    
    \node[loss_node, above=1.2cm of L_init, xshift=1cm, minimum width=6cm, fill=orange!25, font=\bfseries] (L_total) {$L_{total} = \sum \lambda_i L_i$};

    \node[stage_box, fit=(L_sup) (L_reg) (L_total), inner sep=20pt] (Stage5) {};
    \node[stage_label] at (Stage5.south west) {5. Loss Functions (Grounding Pipeline)};

    % ==========================================
    % WIRING (Vertical Flow UPWARDS)
    % ==========================================
    
    % Stage 1 to 2
    \draw[arrow] (I_q) -- (E_q);
    \draw[arrow] (I_r) -- (E_qa);
    \draw[arrow] (I_pid) -- (G_uq);
    \draw[arrow] (I_uid) -- (G_st);

    % Stage 2 details
    \draw[arrow] (E_q) -- (X_prime);
    \draw[arrow] (E_qa) -- (Y_prime);
    \draw[arrow] (G_uq.west) -- ++(-0.25,0) |- (X_prime.east);
    \draw[arrow] (G_uq.east) -- ++(0.25,0) |- (Y_prime.west);
    \draw[arrow] (G_st.west) -- ++(-1.1,0) |- (X_prime.east);
    \draw[arrow] (G_st.north) -- (Y_prime.south);

    % Stage 2 to 3
    \draw[arrow] (X_prime) -- (Decoder);
    \draw[arrow] (Y_prime) -- (Encoder);
    
    % Cross-Attention Path
    \draw[arrow] (Encoder.west) -- ++(-1,0) |- (Decoder.east) node[midway, left, pos=0.8, font=\scriptsize] {Context $y^*$};

    % Stage 3 to 4
    \draw[arrow] (Decoder) -- (MLP);
    \draw[arrow] (MLP) -- (Predict);

    % Stage 4/1 to 5 (Loss Inputs)
    \draw[arrow] (Predict) -- (L_sup);
    \draw[arrow] (Predict) -- (L_ref);
    
    % BKT Highway (Far Right - Flowing UP)
    \draw[path_line] (I_bkt.north) -- ++(0,18) coordinate (bkt_h);
    \draw[arrow] (bkt_h) -| (L_ref.south);
    \draw[arrow] (bkt_h) -| (L_init.south);
    \draw[arrow] (bkt_h) -| (L_rate.south);
    
    % Params Highway (Middle Right - Flowing UP)
    \draw[path_line] (G_st.north) -- ++(0,11) coordinate (param_h);
    \draw[arrow] (param_h) -| (L_init.south);
    \draw[arrow] (param_h) -| (L_rate.south);
    \draw[arrow] (param_h) -| (L_reg.south);
    
    % Rasch Highway
    \draw[path_line] (G_uq.north) -- ++(0,11) coordinate (rasch_h);
    \draw[arrow] (rasch_h) -| (L_rasch.south);
    
    % Total Loss Aggregation
    \foreach \l in {L_sup, L_ref, L_init, L_rate, L_rasch, L_reg}
        \draw[arrow] (\l.north) -- (L_total.south -| \l.north);

\end{tikzpicture}
\end{document}
