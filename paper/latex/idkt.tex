\newcommand{\vectorstack}[4]{% #1: name, #2: color, #3: var_prefix, #4: y_offset
    \begin{scope}[yshift=#4]
        \node[v_cell, fill=#2] (#1_1) at (0,0) {${#3}_{1}$};
        \node[v_cell, fill=#2] (#1_2) at (0.55,0) {${#3}_{2}$};
        \node[font=\tiny] at (1.05,0) {$\dots$};
        \node[v_cell, fill=#2] (#1_3) at (1.55,0) {${#3}_{n-1}$};
        \node[v_cell, fill=#2] (#1_n) at (2.1,0) {${#3}_{n}$};
        \node[invisible, fit=(#1_1) (#1_n)] (#1) {};
    \end{scope}
}
