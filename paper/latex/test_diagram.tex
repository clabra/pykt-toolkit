% Define a simple command for the stacks since \vectorstack isn't standard
\newcommand{\vectorstack}[4]{
    \node[subblock, fill=#2, yshift=#4] (#1) {#3};
    \node[subblock, fill=#2, yshift=#4+0.4cm] (#1_2) {};
}

% --- LAYER 1: INPUT ---
\node[data] (I_pid) {Problem IDs $pid$};
\node[data, right=0.8cm of I_pid] (I_q) {Questions $q$};
\node[data, right=0.8cm of I_q] (I_r) {Responses $r$};
\node[data, right=0.8cm of I_r] (I_uid) {Student IDs $uid$};
\node[ref_node, right=1.2cm of I_uid] (I_bkt) {BKT Reference\\Targets};

\node[stage_box, fit=(I_pid) (I_bkt), inner sep=12pt] (Stage1) {};
\node[stage_label, anchor=north west] at (Stage1.north west) {1. Input Data};

% --- LAYER 2: EMBEDDINGS ---
% x' Stack
\begin{scope}[shift={($(I_q.north)+(0,2.5)$)}]
    \vectorstack{X_final}{blue!20}{x'}{0.65cm}
    \vectorstack{X_ground}{blue!10}{l_c}{0cm}
    \vectorstack{X_base}{blue!5}{c}{-0.8cm}
    \node at (0, -0.4) {$\ominus$};
\end{scope}

% y' Stack
\begin{scope}[shift={($(I_r.north)+(0,2.5)$)}]
    \vectorstack{Y_final}{orange!20}{y'}{0.65cm}
    \vectorstack{Y_ground}{orange!10}{t_c}{0cm}
    \vectorstack{Y_base}{orange!5}{e}{-0.8cm}
    \node at (0, -0.4) {$\oplus$};
\end{scope}

\node[grounding, above=3.5cm of I_pid] (G_uq) {Question\\Difficulty ($u_q$)};
\node[grounding, above=3.5cm of I_uid] (G_st) {Student\\Params ($k_c, v_s$)};

\node[stage_box, fit=(G_uq) (G_st) (X_final_2) (Y_final_2), inner sep=12pt] (Stage2) {};
\node[stage_label, anchor=north west] at (Stage2.north west) {2. Embeddings};

% --- LAYER 3: CORE ---
\node[attn_sub, above=4cm of Y_final_2] (E_MHA) {Multi-Head Self-Attn};
\node[norm_sub, above=0.3cm of E_MHA] (E_AN1) {Add \& Norm};
\node[ff_sub, above=0.4cm of E_AN1] (E_FFN) {Feed Forward};

\node[stage_box, fit=(E_MHA) (E_AN2), inner sep=5pt] (EncBlock) {};

\node[attn_sub, above=3cm of X_final_2] (D_MHA) {Masked MHA};
\node[norm_sub, above=0.25cm of D_MHA] (D_AN1) {Add \& Norm};
\node[attn_sub, above=0.4cm of D_AN1] (D_CA) {Cross-Attn};

\node[stage_box, fit=(D_MHA) (D_AN2), inner sep=5pt] (DecBlock) {};

\node[output, above=0.7cm of MLP] (Predict) {iDKT Prediction};

\node[loss_node, above=2cm of Predict, xshift=-2cm] (L_sup) {$L_{sup}$};
\node[loss_node, right=0.4cm of L_sup] (L_ref) {$L_{ref}$};
